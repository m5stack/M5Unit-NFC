54inline void pack_le24(uint8_t out[3], const uint32_t value)
55{
56 out[0] = static_cast<uint8_t>(value & 0xFF);
57 out[1] = static_cast<uint8_t>((value >> 8) & 0xFF);
58 out[2] = static_cast<uint8_t>((value >> 16) & 0xFF);
59}
60
61inline void pack_be24(uint8_t out[3], const uint32_t value)
62{
63 out[0] = static_cast<uint8_t>((value >> 16) & 0xFF);
64 out[1] = static_cast<uint8_t>((value >> 8) & 0xFF);
65 out[2] = static_cast<uint8_t>(value & 0xFF);
66}
67
68inline uint32_t unpack_le24(const uint8_t in[3])
69{
70 return static_cast<uint32_t>(in[0]) | (static_cast<uint32_t>(in[1]) << 8) | (static_cast<uint32_t>(in[2]) << 16);
71}
